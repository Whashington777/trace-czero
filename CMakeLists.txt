cmake_minimum_required(VERSION 3.20)
project(tracecs16 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-m32)
add_link_options(-m32)
set(PORTFOLIO_32BIT ON CACHE BOOL "" FORCE)
set(PORTFOLIO_BUILD_PREVIEW OFF CACHE BOOL "" FORCE)
set(GUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/Imgui Portifolio #8")
add_subdirectory("${GUI_DIR}" gui)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2_HEADERS REQUIRED sdl2)
set(SDL2_RUNTIME "$ENV{HOME}/.local/share/Steam/ubuntu12_32/steam-runtime/usr/lib/i386-linux-gnu/libSDL2-2.0.so.0" CACHE FILEPATH "32-bit SDL2 runtime")
if(NOT EXISTS "${SDL2_RUNTIME}")
 message(FATAL_ERROR "Set SDL2_RUNTIME to an installed 32-bit libSDL2-2.0.so.0")
endif()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/runtime")
configure_file("${SDL2_RUNTIME}" "${CMAKE_BINARY_DIR}/runtime/libSDL2-2.0.so.0" COPYONLY)
set(SDL2_LOCAL "${CMAKE_BINARY_DIR}/runtime/libSDL2-2.0.so.0")
set(IMGUI_DIR "${GUI_DIR}/thirdparty/imgui-current")
add_library(tracecs16 SHARED src/base/modules.cpp src/base/game.cpp src/base/runtime.cpp
 "${IMGUI_DIR}/backends/imgui_impl_sdl2.cpp" "${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp")
target_include_directories(tracecs16 PRIVATE src ${SDL2_HEADERS_INCLUDE_DIRS})
target_link_libraries(tracecs16 PRIVATE portfolio_gui "${SDL2_LOCAL}" dl)
target_compile_options(tracecs16 PRIVATE -Wall -Wextra)
target_link_options(tracecs16 PRIVATE -Wl,--no-undefined)
set_target_properties(tracecs16 PROPERTIES BUILD_RPATH "$ORIGIN/gui")
include(CTest)
add_executable(base_math_tests tests/base_math.cpp)
target_include_directories(base_math_tests PRIVATE src)
add_test(NAME base_math COMMAND base_math_tests)
add_executable(overlay_host tests/overlay_host.cpp)
target_include_directories(overlay_host PRIVATE ${SDL2_HEADERS_INCLUDE_DIRS})
target_link_libraries(overlay_host PRIVATE "${SDL2_LOCAL}" GL dl)
add_test(NAME overlay_host COMMAND ${CMAKE_COMMAND} -E env "LD_PRELOAD=$<TARGET_FILE:tracecs16>" $<TARGET_FILE:overlay_host>)
add_executable(aim_selection_tests tests/aim_selection.cpp src/base/game.cpp src/base/modules.cpp)
target_include_directories(aim_selection_tests PRIVATE src)
target_link_libraries(aim_selection_tests PRIVATE portfolio_gui dl)
add_test(NAME aim_selection COMMAND aim_selection_tests)

add_test(NAME overlay_detach COMMAND ${CMAKE_COMMAND} -E env "LD_PRELOAD=$<TARGET_FILE:tracecs16>" $<TARGET_FILE:overlay_host> --detach)
add_executable(snapshot_tests tests/snapshot.cpp src/base/game.cpp src/base/modules.cpp)
target_include_directories(snapshot_tests PRIVATE src)
target_link_libraries(snapshot_tests PRIVATE portfolio_gui dl)
add_test(NAME snapshot COMMAND snapshot_tests)
add_executable(trace_launcher src/launcher/main.cpp
 "${IMGUI_DIR}/backends/imgui_impl_sdl2.cpp" "${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp")
target_compile_definitions(trace_launcher PRIVATE _GNU_SOURCE)
target_include_directories(trace_launcher PRIVATE ${SDL2_HEADERS_INCLUDE_DIRS})
target_link_libraries(trace_launcher PRIVATE portfolio_gui "${SDL2_LOCAL}" GL)

set_target_properties(trace_launcher PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/gui;$ORIGIN/runtime")
set_target_properties(tracecs16 PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/gui;$ORIGIN/runtime")
set_target_properties(portfolio_gui PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/../runtime")
